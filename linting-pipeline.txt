 name: Linting Check

on:
  push:
    branches: [develop]
  pull_request:
    branches: [develop]

jobs:
  lint:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: '18'

      - name: Install dependencies
        run: npm ci

      - name: Run HTMLHint
        run: npx htmlhint "**/*.html" || true

      - name: Run Stylelint
        run: npx stylelint "**/*.css" || true
